\begin{frame}{Dados Públicos}
	Uso dos seus dados por pessoas má intencionadas:
	\begin{center}
		\begin{itemize}
			\item Invasão de residências
			\item Sequestros
			\item Exposição de crianças para pedófilos
		\end{itemize}
	\end{center}
\end{frame}

\begin{frame}{Phishing}
	Atacante busca se passar por algum site conhecido. Principais perigos:
	\begin{itemize}
		\item Baixar arquivos
		\item Clicar em botão
		\item Inserir os dados
	\end{itemize}
\end{frame}

\begin{frame}{Engenharia Social}
	Atacante busca manipular as emoções da vitima. Exemplos:
	\begin{itemize}
		\item "Sua conta será bloqueada em 1h, entre nesse link"
		\item "Sou do suporte preciso do acesso remoto"
		\item "Pai, troquei de número, me passa um Pix"
	\end{itemize}
\end{frame}

\begin{frame}{Dados comprometidos}
	\begin{itemize}
		\item Clonagem de chip
		\item Boleto de fatura falso
		\item Rastreamento de compra falso
	\end{itemize}
\end{frame}

\begin{frame}{Uso dos seus dados}
	\begin{itemize}
		\item Empresa Fantama
		\item Conta falsa (Uber)
		\item Crediário em loja
		\item Empréstimo
		\item Financiamento
		\item Cartão de crédito
	\end{itemize}
\end{frame}
